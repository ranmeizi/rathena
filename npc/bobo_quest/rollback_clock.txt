// 钟怪击杀计数（示例，需配合mob杀怪脚本）
-	script	钟怪计数#kill	-1,{
    OnNPCKillEvent:
    if (quest_clock == 3 && killedrid == 1268) { // 1268为钟怪ID
        set clock_kill, clock_kill + 1;
        if (clock_kill >= 30) {
            dispbottom "你已经击杀了30个钟怪，回去找钟匠吧！";
            set quest_clock,4;
        }
    }
    end;
}
